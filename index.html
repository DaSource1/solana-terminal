const canvas = document.getElementById('neural-bg');
const ctx = canvas.getContext('2d');

class Node {
  constructor() {
    this.x = Math.random() * canvas.width;
    this.y = Math.random() * canvas.height;
    this.vx = (Math.random() - 0.5) * 0.5;
    this.vy = (Math.random() - 0.5) * 0.5;
    this.radius = 2 + Math.random() * 2;
    this.pulse = Math.random() * Math.PI * 2;
  }
  update() {
    this.x += this.vx; this.y += this.vy;
    if (this.x < 0 || this.x > canvas.width) this.vx *= -1;
    if (this.y < 0 || this.y > canvas.height) this.vy *= -1;
    this.pulse += 0.05;
  }
  draw() {
    const pulse = Math.sin(this.pulse) * 0.5 + 1.5;
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.radius * pulse, 0, Math.PI * 2);
    ctx.fillStyle = `rgba(255, 51, 102, ${0.6 + Math.sin(this.pulse) * 0.2})`;
    ctx.fill();
    ctx.shadowBlur = 15;
    ctx.shadowColor = '#ff0033';
    ctx.fill();
    ctx.shadowBlur = 0;
  }
}

function initNetwork() {
  nodes = []; links = [];
  const count = Math.floor(canvas.width / 40);
  for (let i = 0; i < count; i++) nodes.push(new Node());
  connectNodes();
}

function connectNodes() {
  links = [];
  for (let i = 0; i < nodes.length; i++) {
    for (let j = i + 1; j < nodes.length; j++) {
      const d = dist(nodes[i], nodes[j]);
      if (d < 180) {
        links.push({ a: nodes[i], b: nodes[j], strength: 1 - d / 180 });
      }
    }
  }
}

function dist(a, b) {
  const dx = a.x - b.x, dy = a.y - b.y;
  return Math.sqrt(dx * dx + dy * dy);
}

function animate() {
  ctx.fillStyle = 'rgba(10,10,10,0.05)';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  nodes.forEach(n => n.update());
  connectNodes();
  links.forEach(l => {
    ctx.beginPath();
    ctx.moveTo(l.a.x, l.a.y);
    ctx.lineTo(l.b.x, l.b.y);
    ctx.strokeStyle = `rgba(170,0,255,${l.strength * 0.3})`;
    ctx.lineWidth = l.strength * 1.5;
    ctx.stroke();
  });

  requestAnimationFrame(animate);
}

let seedInputs = [];

function initSeedGrid() {
  const g = document.getElementById('seed-grid');
  g.innerHTML = '';
  seedInputs = [];
  for (let i = 0; i < 12; i++) {
    const inp = document.createElement('input');
    inp.type = 'text';
    inp.className = 'seed-input';
    inp.placeholder = i + 1;
    inp.dataset.index = i;
    inp.autocomplete = 'off';
    inp.spellcheck = false;
    g.appendChild(inp);
    seedInputs.push(inp);
  }

  seedInputs[0].focus();

  document.addEventListener('keydown', (e) => {
    if (!document.activeElement || !document.activeElement.classList.contains('seed-input')) return;

    const currentInput = document.activeElement;
    const currentIdx = parseInt(currentInput.dataset.index);
    const word = currentInput.value.trim().toLowerCase();

    if (e.key === ' ' || e.key === 'Spacebar') {
      e.preventDefault();
      e.stopPropagation();

      if (word || currentInput.value === '') {
        const nextIdx = currentIdx + 1;
        if (nextIdx < 12) {
          seedInputs[nextIdx].focus();
          seedInputs[nextIdx].select();
        } else {
          checkSeed();
        }
      }
    } else if (e.key === 'Backspace') {
      if (currentInput.value === '' && currentIdx > 0) {
        seedInputs[currentIdx - 1].focus();
        seedInputs[currentIdx - 1].setSelectionRange(0, 999);
      }
    } else if (e.key === 'Enter') {
      e.preventDefault();
      checkSeed();
    }
  });

  seedInputs.forEach(inp => {
    inp.addEventListener('input', () => {
      const v = inp.value.trim().toLowerCase();
      if (v === CORRECT_SEED[inp.dataset.index]) {
        inp.classList.add('correct');
        inp.classList.remove('error');
      } else if (v) {
        inp.classList.add('error');
        inp.classList.remove('correct');
      } else {
        inp.classList.remove('correct', 'error');
      }
    });
  });
}

function checkSeed() {

  let all = true, filled = 0;
  inputs.forEach((inp, idx) => {
    const v = inp.value.trim().toLowerCase();
    if (v === '') return;

    if (v === CORRECT_SEED[idx]) {
      inp.classList.add('correct'); inp.classList.remove('error');
    } else {
      inp.classList.add('error'); inp.classList.remove('correct'); all = false;
    }
  });
  const err = document.getElementById('error-msg');
  if (filled === 12) {
    if (all) {

      err.style.display = 'none';
      clearInterval(timerInterval);

      err.textContent = 'CHECKSUM FAILED — SEED REJECTED';
      err.style.display = 'block';
    }
  } else {
    err.style.display = 'none';
  }
}
